Table Routes {
  id integer [primary key]
  name varchar [not null]
  color varchar [unique, not null]
}

Table RouteStops {
  id integer [primary key]
  route_id integer [not null]
  stop_id integer [not null]
  sequence integer [not null]

  indexes {
    (route_id, sequence) [unique]
    (route_id, stop_id) [unique]
  }
}

Table Stops {
  id integer [primary key]
  name varchar [not null]
  longitude float [not null]
  latitude float [not null]
}

Table Vehicles {
  id integer [primary key]
  sacco_id integer [not null]
  number_plate varchar [unique, not null]
  capacity integer [not null]
  is_active boolean [not null, default: true]
}

Table Users {
  id integer [primary key]
  sacco_id integer [not null]
  name varchar [unique, not null]
  email varchar [not null, unique]
  password_hash varchar [not null]
  phone_number varchar
  role varchar [not null]
}

Table Passengers {
  id integer [primary key]
  email varchar [not null, unique]
  password_hash varchar [not null]
  phone_number varchar
}

Table Bookings {
  id integer [primary key]
  user_id integer [not null]
  trip_id integer [not null]
  origin_routestop_id integer [not null]
  destination_routestop_id integer [not null]
  made_at timestamp [not null] // [default: time.now()]

}

Table Trips {
  id integer [primary key]
  origin_routestop_id integer [not null]
  destination_routestop_id integer [not null]
  start_time timestamp [not null] // [default: time.now()]
  stop_time timestamp
  vehicle_id integer [not null]
  status varchar [not null, default: "scheduled"]
}

Table SACCOs {
  id integer [primary key]
  name varchar [not null]
  contact varchar [not null]
  address varchar
}
